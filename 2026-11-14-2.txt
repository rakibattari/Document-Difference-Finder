        function loadSampleData() {

            compareTexts();
        }

        function clearText(elementId) {
            document.getElementById(elementId).value = '';
            compareTexts();
        }

        function compareTexts() {
            const original = document.getElementById('originalText').value;
            const modified = document.getElementById('modifiedText').value;
            const outputDiv = document.getElementById('diffOutput');

            if (!original && !modified) {
                outputDiv.innerHTML = '<span class="text-slate-500">ফলাফল দেখতে উপরে টেক্সট ইনপুট দিন...</span>';
                return;
            }

            const diff = Diff.diffWordsWithSpace(original, modified);
            let fragment = '';

            diff.forEach(part => {
                if (part.added) {
                    fragment += `<span class="ins">${escapeHtml(part.value)}</span>`;
                } else if (part.removed) {
                    fragment += `<span class="del">${escapeHtml(part.value)}</span>`;
                } else {
                    fragment += `<span>${escapeHtml(part.value)}</span>`;
                }
            });

            outputDiv.innerHTML = fragment;
        }

        function escapeHtml(text) {
            return text
                .replace(/&/g, "&amp;")
                .replace(/</g, "&lt;")
                .replace(/>/g, "&gt;")
                .replace(/"/g, "&quot;")
                .replace(/'/g, "&#039;");
        }
